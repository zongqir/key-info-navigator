# 思源笔记备注提取API - 实用指南

## 🎯 核心提取函数

### 1. 获取页面所有备注
```typescript
/**
 * 获取指定页面的所有备注
 * @param mainNode 编辑器主节点 (通常是 .protyle-wysiwyg 元素)
 * @returns 备注数组
 */
function getAllMemos(mainNode: HTMLElement) {
    // 查找所有备注元素
    const memoElements = mainNode.querySelectorAll("span[data-type*='inline-memo']");
    
    const memos = [];
    memoElements.forEach((element, index) => {
        const memo = {
            // 基本信息
            id: index,
            element: element,
            
            // 内容信息
            displayText: element.textContent,                              // 显示的文本
            memoContent: element.getAttribute("data-inline-memo-content"), // 备注内容
            
            // 位置信息
            blockElement: findParentBlock(element),                        // 所在块元素
            blockId: findParentBlock(element)?.dataset.nodeId,            // 块ID
            
            // 位置坐标
            position: {
                top: element.getBoundingClientRect().top,
                left: element.getBoundingClientRect().left
            }
        };
        memos.push(memo);
    });
    
    return memos;
}
```

### 2. 查找备注所在的块
```typescript
/**
 * 向上查找备注所在的块元素
 * @param memoElement 备注元素
 * @returns 包含该备注的块元素
 */
function findParentBlock(memoElement: HTMLElement): HTMLElement | null {
    let current = memoElement;
    
    // 向上遍历，直到找到有 data-node-id 的元素
    while (current && current.parentElement) {
        if (current.dataset.nodeId) {
            return current;
        }
        current = current.parentElement;
    }
    
    return null;
}
```

### 3. 按块分组备注
```typescript
/**
 * 将备注按所在块进行分组
 * @param memos 备注数组
 * @returns 按块ID分组的备注对象
 */
function groupMemosByBlock(memos: any[]) {
    const grouped = {};
    
    memos.forEach(memo => {
        const blockId = memo.blockId;
        if (!grouped[blockId]) {
            grouped[blockId] = {
                blockId: blockId,
                blockElement: memo.blockElement,
                memos: []
            };
        }
        grouped[blockId].memos.push(memo);
    });
    
    return grouped;
}
```

### 4. 合并相邻相同备注
```typescript
/**
 * 合并相邻的相同备注（思源的特殊处理）
 * @param memos 原始备注数组
 * @returns 合并后的备注数组
 */
function mergeSimilarMemos(memos: any[]) {
    const merged = [];
    
    memos.forEach((memo, index) => {
        if (index === 0) {
            // 第一个备注直接添加
            merged.push({
                ...memo,
                mergedDisplayText: memo.displayText,
                originalIndexes: [index]
            });
        } else {
            const lastMerged = merged[merged.length - 1];
            const lastOriginalIndex = lastMerged.originalIndexes[lastMerged.originalIndexes.length - 1];
            const lastOriginalMemo = memos[lastOriginalIndex];
            
            // 检查是否需要合并：相邻 + 同块 + 同内容
            const isAdjacent = lastOriginalMemo.element.nextSibling === memo.element;
            const sameBlock = lastOriginalMemo.blockId === memo.blockId;
            const sameContent = lastOriginalMemo.memoContent === memo.memoContent;
            
            if (isAdjacent && sameBlock && sameContent) {
                // 合并到上一个
                lastMerged.mergedDisplayText += memo.displayText;
                lastMerged.originalIndexes.push(index);
            } else {
                // 创建新的合并项
                merged.push({
                    ...memo,
                    mergedDisplayText: memo.displayText,
                    originalIndexes: [index]
                });
            }
        }
    });
    
    return merged;
}
```

## 🚀 完整使用示例

### 基础使用
```typescript
// 1. 获取编辑器主节点
const mainNode = document.querySelector("div.protyle-wysiwyg");

// 2. 提取所有备注
const allMemos = getAllMemos(mainNode);

// 3. 打印备注信息
allMemos.forEach((memo, index) => {
    console.log(`备注 ${index + 1}:`);
    console.log(`  显示文本: ${memo.displayText}`);
    console.log(`  备注内容: ${memo.memoContent}`);
    console.log(`  所在块ID: ${memo.blockId}`);
    console.log('---');
});
```

### 高级使用 - 按块分组
```typescript
// 获取所有备注
const allMemos = getAllMemos(mainNode);

// 按块分组
const groupedMemos = groupMemosByBlock(allMemos);

// 遍历每个块的备注
Object.values(groupedMemos).forEach(blockGroup => {
    console.log(`块 ${blockGroup.blockId} 包含 ${blockGroup.memos.length} 个备注:`);
    blockGroup.memos.forEach(memo => {
        console.log(`  - ${memo.displayText}: ${memo.memoContent}`);
    });
});
```

### 高级使用 - 合并相似备注
```typescript
// 获取并合并备注
const allMemos = getAllMemos(mainNode);
const mergedMemos = mergeSimilarMemos(allMemos);

// 显示合并后的结果
mergedMemos.forEach((memo, index) => {
    console.log(`合并备注 ${index + 1}:`);
    console.log(`  合并文本: ${memo.mergedDisplayText}`);
    console.log(`  备注内容: ${memo.memoContent}`);
    console.log(`  原始索引: ${memo.originalIndexes.join(', ')}`);
    console.log('---');
});
```

## 📊 数据结构说明

### 单个备注对象
```typescript
interface Memo {
    id: number;                    // 备注序号
    element: HTMLElement;          // DOM元素
    displayText: string;           // 显示文本（用户看到的）
    memoContent: string;           // 备注内容（实际注释）
    blockElement: HTMLElement;     // 所在块元素
    blockId: string;              // 块唯一ID
    position: {                   // 屏幕位置
        top: number;
        left: number;
    };
}
```

### 合并后的备注对象
```typescript
interface MergedMemo extends Memo {
    mergedDisplayText: string;     // 合并后的显示文本
    originalIndexes: number[];     // 原始备注的索引数组
}
```

### 分组后的块对象
```typescript
interface BlockGroup {
    blockId: string;              // 块ID
    blockElement: HTMLElement;    // 块DOM元素
    memos: Memo[];               // 该块内的所有备注
}
```

## 🔧 实用工具函数

### 监听备注变化
```typescript
/**
 * 监听备注内容变化
 * @param callback 变化时的回调函数
 */
function watchMemoChanges(callback: (changedMemo: any) => void) {
    const observer = new MutationObserver((mutations) => {
        mutations.forEach((mutation) => {
            if (mutation.type === 'attributes' && 
                mutation.attributeName === 'data-inline-memo-content') {
                
                const element = mutation.target as HTMLElement;
                const memo = {
                    element: element,
                    displayText: element.textContent,
                    memoContent: element.getAttribute("data-inline-memo-content"),
                    blockId: findParentBlock(element)?.dataset.nodeId
                };
                
                callback(memo);
            }
        });
    });
    
    // 监听所有备注元素
    const memoElements = document.querySelectorAll("span[data-type*='inline-memo']");
    memoElements.forEach(element => {
        observer.observe(element, { attributes: true });
    });
    
    return observer; // 返回observer以便后续取消监听
}
```

### 查找特定内容的备注
```typescript
/**
 * 查找包含特定内容的备注
 * @param searchText 搜索文本
 * @param mainNode 编辑器节点
 * @returns 匹配的备注数组
 */
function findMemosByContent(searchText: string, mainNode: HTMLElement) {
    const allMemos = getAllMemos(mainNode);
    return allMemos.filter(memo => 
        memo.memoContent.includes(searchText) || 
        memo.displayText.includes(searchText)
    );
}
```

## 💡 使用场景示例

### 1. 导出所有备注到文本
```typescript
function exportMemosToText(mainNode: HTMLElement): string {
    const memos = getAllMemos(mainNode);
    const grouped = groupMemosByBlock(memos);
    
    let result = "# 页面备注导出\n\n";
    
    Object.values(grouped).forEach(blockGroup => {
        result += `## 块 ${blockGroup.blockId}\n`;
        blockGroup.memos.forEach((memo, index) => {
            result += `${index + 1}. **${memo.displayText}**\n`;
            result += `   备注: ${memo.memoContent}\n\n`;
        });
    });
    
    return result;
}
```

### 2. 统计备注数量
```typescript
function getMemoStats(mainNode: HTMLElement) {
    const memos = getAllMemos(mainNode);
    const grouped = groupMemosByBlock(memos);
    
    return {
        totalMemos: memos.length,
        totalBlocks: Object.keys(grouped).length,
        averageMemosPerBlock: memos.length / Object.keys(grouped).length,
        blockStats: Object.values(grouped).map(block => ({
            blockId: block.blockId,
            memoCount: block.memos.length
        }))
    };
}
```
